const app = qq.ce({ tag: 'div', style: { position: 'absolute', left: '20px', top: '20px', width: '240px', minHeight: '150px', zIndex: '5', background: 'lightblue', padding: '10px' } }) const out = qq.ce({ tag: 'div', parent: app, style: { whiteSpace: 'pre-wrap' } }) const ns = qq.ns() const queue = [] let busy = false function execute(message) { out.innerText += message + '\n' busy = true setTimeout(() => { busy = false next() }, 3000) } function next() { if (busy || !queue.length) return execute(queue.shift()) } function dispatch(message) { queue.push(message) next() } ns.subscribe('dispatch', dispatch) qq.ce({ tag: 'button', parent: app, it: 'put', data: { id: 'put' }, event: { click: function() { ns.set('dispatch', this.data.id) } } }) setTimeout(() => ns.set('dispatch', '5'), 5000) setTimeout(() => ns.set('dispatch', '10'), 10000)
